#     python -m venv .venv
#     .venv\Scripts\activate        (Windows)
#     source .venv/bin/activate     (Linux / macOS)
#     pip install -r requirements.txt
#
# Solo se listan las dependencias DIRECTAS del proyecto. Las transitivas

# repetirlas aquí solo crearía dos sitios que mantener sincronizados.
#
# Las versiones van fijadas con == para que todos los equipos levanten
# exactamente el mismo entorno: un rango abierto haría que quien instale
